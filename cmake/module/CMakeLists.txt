cmake_minimum_required(VERSION 3.18)
project(phoenix_module LANGUAGES NONE)
get_filename_component(PHXFS_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(module_root "${PHXFS_ROOT}/module")
set(module_output "${PHXFS_OUTPUT_DIR}/module")
set(VENDOR_KERNEL_MODULE "")
string(TOLOWER "${PHXFS_VENDOR}" vendor_name)
set(VENDOR_BACKEND "${vendor_name}-backend.o")
set(VENDOR_DEFINE "CONFIG_PHXFS_VENDOR_${PHXFS_VENDOR}")
if(PHXFS_VENDOR STREQUAL "METAX")
    set(VENDOR_KERNEL_MODULE "/opt/mxdriver/kernel_module/${CMAKE_SYSTEM_VERSION}/Module.symvers")
endif()
if(NOT EXISTS "${module_root}/${vendor_name}-backend.c")
    message(FATAL_ERROR "Kernel backend is not implemented: ${PHXFS_VENDOR}")
endif()

set(KERNEL "/lib/modules/${CMAKE_SYSTEM_VERSION}/build" CACHE PATH "Kernel build tree")
if(NOT EXISTS "${KERNEL}/Makefile")
    message(FATAL_ERROR "Kernel build tree missing: ${KERNEL}. Install kernel-devel matching ${CMAKE_SYSTEM_VERSION}.")
endif()
set(driver_include "")
if(PHXFS_VENDOR STREQUAL "NVIDIA")
    set(NVIDIA "" CACHE PATH "NVIDIA driver header directory")
    if(NVIDIA)
        foreach(dir "${NVIDIA}" "${NVIDIA}/nvidia-peermem" "${NVIDIA}/kernel/nvidia" "${NVIDIA}/kernel-open/nvidia")
            if(EXISTS "${dir}/nv-p2p.h")
                set(driver_include "${dir}")
                break()
            endif()
        endforeach()
    else()
        set(driver_version "")
        if(EXISTS "/proc/driver/nvidia/version")
            file(READ "/proc/driver/nvidia/version" driver_info)
            string(REGEX MATCH "Kernel Module[ \t]+([0-9]+[.][0-9]+([.][0-9]+)?)" driver_match "${driver_info}")
            set(driver_version "${CMAKE_MATCH_1}")
        endif()
        if(driver_version)
            foreach(dir "${PHXFS_EXTRA_HEADERS}/nvidia-${driver_version}"
                    "/usr/src/nvidia-${driver_version}/nvidia-peermem"
                    "/usr/src/nvidia-${driver_version}/kernel/nvidia"
                    "/usr/src/nvidia-${driver_version}/kernel-open/nvidia"
                    "/usr/src/nvidia-${driver_version}")
                if(EXISTS "${dir}/nv-p2p.h")
                    set(driver_include "${dir}")
                    break()
                endif()
            endforeach()
        endif()
        if(NOT driver_include AND EXISTS "${PHXFS_EXTRA_HEADERS}/nv-p2p.h")
            set(driver_include "${PHXFS_EXTRA_HEADERS}")
        endif()
        if(NOT driver_include)
            string(REGEX MATCH "^[0-9]+" driver_branch "${driver_version}")
            foreach(version 535.129.03 550.54.15 570.124.06 575.57.08 580.95.05 590.44.01)
                string(REGEX MATCH "^[0-9]+" header_branch "${version}")
                if(driver_branch STREQUAL header_branch AND
                        EXISTS "${PHXFS_ROOT}/extra_headers/nvidia-${version}/nv-p2p.h")
                    set(driver_include "${PHXFS_ROOT}/extra_headers/nvidia-${version}")
                    break()
                endif()
            endforeach()
        endif()
    endif()
    if(NOT driver_include)
        message(FATAL_ERROR "nv-p2p.h missing for NVIDIA driver ${driver_version}. Put it in ${PHXFS_EXTRA_HEADERS}/nvidia-${driver_version}/, or set NVIDIA to its directory and run cmake again.")
    endif()
    message(STATUS "NVIDIA P2P header: ${driver_include}/nv-p2p.h")
endif()
set(module_ccflags "-D${VENDOR_DEFINE}")
if(driver_include)
    string(APPEND module_ccflags " -I${driver_include}")
endif()
if(PHXFS_MAP_MODE STREQUAL "full")
    set(PHXFS_MAP_MODE_DEFAULT 0)
else()
    set(PHXFS_MAP_MODE_DEFAULT 1)
endif()
string(APPEND module_ccflags " -DPHXFS_MAP_MODE_DEFAULT=${PHXFS_MAP_MODE_DEFAULT}")
configure_file("${module_root}/Makefile.in" "${module_output}/Makefile" @ONLY)
