cmake_minimum_required(VERSION 3.18)
get_filename_component(PHXFS_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

if(PHXFS_VENDOR STREQUAL "METAX")
    include("${PHXFS_ROOT}/cmake/MetaX.cmake")
endif()
project(phoenix_library LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)
if(PHXFS_VENDOR STREQUAL "NVIDIA")
    find_package(CUDAToolkit 12.4 REQUIRED)
endif()
option(no_liburing "Disable the io_uring engine" OFF)
option(phx_nvtx "Emit NVTX ranges from libphoenix" ON)
option(BUILD_SHARED_LIBS "Build a shared user library" ON)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND NOT no_liburing)
    pkg_check_modules(LIBURING IMPORTED_TARGET liburing)
endif()

set(libphoenix_root "${PHXFS_ROOT}/libphoenix")
file(GLOB sources CONFIGURE_DEPENDS "${libphoenix_root}/*.cpp" "${libphoenix_root}/io_engine/*.cpp")
file(GLOB headers CONFIGURE_DEPENDS "${libphoenix_root}/include/*.h")
if(NOT LIBURING_FOUND OR no_liburing)
    list(REMOVE_ITEM sources "${libphoenix_root}/io_engine/io_engine_uring.cpp")
endif()
if(PHXFS_VENDOR STREQUAL "NVIDIA" OR PHXFS_VENDOR STREQUAL "METAX" OR
        PHXFS_VENDOR STREQUAL "ILUVATAR")
    list(APPEND sources "${libphoenix_root}/connectors/nvidia_connector.cpp")
    if(PHXFS_VENDOR STREQUAL "ILUVATAR")
        list(APPEND sources "${libphoenix_root}/connectors/iluvatar_cuda_rt.cpp")
    endif()
elseif(PHXFS_VENDOR STREQUAL "AMD")
    list(APPEND sources "${libphoenix_root}/connectors/amd_connector.cpp")
elseif(PHXFS_VENDOR STREQUAL "HUAWEI")
    list(APPEND sources "${libphoenix_root}/connectors/huawei_connector.cc")
endif()

add_library(libphoenix ${sources} ${headers})
set_target_properties(libphoenix PROPERTIES OUTPUT_NAME phoenix
    LIBRARY_OUTPUT_DIRECTORY "${PHXFS_OUTPUT_DIR}"
    ARCHIVE_OUTPUT_DIRECTORY "${PHXFS_OUTPUT_DIR}")
target_include_directories(libphoenix PUBLIC "${libphoenix_root}/include" PRIVATE "${libphoenix_root}")
target_compile_options(libphoenix PRIVATE -Wall -Wextra)
target_link_libraries(libphoenix PUBLIC Threads::Threads PRIVATE ${CMAKE_DL_LIBS})
if(no_cc)
    target_compile_definitions(libphoenix PRIVATE __NO_COHERENCE__)
endif()
if(PHXFS_VENDOR STREQUAL "NVIDIA")
    target_link_libraries(libphoenix PRIVATE CUDA::cudart)
    if(phx_nvtx)
        target_compile_definitions(libphoenix PRIVATE PHX_NVTX=1)
    endif()
elseif(PHXFS_VENDOR STREQUAL "ILUVATAR")
    # CoreX runtime symbols are resolved by iluvatar_cuda_rt.cpp at runtime.
    set(COREX_PATH "/usr/local/corex" CACHE PATH "CoreX SDK prefix")
    target_include_directories(libphoenix BEFORE PRIVATE "${COREX_PATH}/include")
    target_compile_definitions(libphoenix PRIVATE PHXFS_ILUVATAR=1)
elseif(PHXFS_VENDOR STREQUAL "AMD")
    set(HIP_PATH "/opt/rocm" CACHE PATH "HIP installation prefix")
    target_include_directories(libphoenix PRIVATE "${HIP_PATH}/include")
    target_link_directories(libphoenix PRIVATE "${HIP_PATH}/lib")
    target_link_libraries(libphoenix PRIVATE amdhip64)
endif()
if(LIBURING_FOUND AND NOT no_liburing)
    target_compile_definitions(libphoenix PRIVATE PHXFS_HAVE_LIBURING)
    target_link_libraries(libphoenix PRIVATE PkgConfig::LIBURING)
    message(STATUS "libphoenix: io_uring enabled")
else()
    message(STATUS "libphoenix: io_uring unavailable; using the sync engine")
endif()
install(FILES ${headers} DESTINATION include)
install(TARGETS libphoenix DESTINATION lib)
