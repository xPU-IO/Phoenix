cmake_minimum_required(VERSION 3.18)
get_filename_component(PHXFS_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
if(PHXFS_VENDOR STREQUAL "METAX")
    include("${PHXFS_ROOT}/cmake/MetaX.cmake")
endif()
if(PHXFS_TEST_KIND STREQUAL "gpu" AND
        (PHXFS_VENDOR STREQUAL "NVIDIA" OR PHXFS_VENDOR STREQUAL "METAX"))
    if(NOT CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 90)
    endif()
    project(phoenix_gpu_tests LANGUAGES C CXX CUDA)
    set(CMAKE_CUDA_STANDARD 20)
else()
    project(phoenix_cpu_tests LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PHXFS_OUTPUT_DIR}/bin")
find_package(Threads REQUIRED)
if(PHXFS_VENDOR STREQUAL "NVIDIA")
    find_package(CUDAToolkit 12.4 REQUIRED)
elseif(PHXFS_VENDOR STREQUAL "ILUVATAR")
    set(COREX_PATH "/usr/local/corex" CACHE PATH "CoreX SDK prefix")
endif()
set(phoenix_include "${PHXFS_ROOT}/libphoenix/include")
set(TEST_FILE "/mnt/nvme0/phxfs_test.bin" CACHE STRING "Test I/O file")
add_library(libphoenix UNKNOWN IMPORTED)
set_target_properties(libphoenix PROPERTIES
    IMPORTED_LOCATION "${PHXFS_LIBRARY_FILE}"
    INTERFACE_INCLUDE_DIRECTORIES "${phoenix_include}")
set_property(TARGET libphoenix APPEND PROPERTY INTERFACE_LINK_LIBRARIES
    Threads::Threads ${CMAKE_DL_LIBS})
if(PHXFS_VENDOR STREQUAL "NVIDIA")
    set_property(TARGET libphoenix APPEND PROPERTY INTERFACE_LINK_LIBRARIES CUDA::cudart)
endif()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND NOT no_liburing)
    pkg_check_modules(LIBURING IMPORTED_TARGET liburing)
    if(LIBURING_FOUND)
        set_property(TARGET libphoenix APPEND PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::LIBURING)
    endif()
endif()
if(PHXFS_TEST_KIND STREQUAL "gpu")
    if(PHXFS_VENDOR STREQUAL "AMD")
        add_subdirectory("${PHXFS_ROOT}/test/amd" gpu)
    elseif(PHXFS_VENDOR STREQUAL "ILUVATAR")
        enable_testing()
        add_subdirectory("${PHXFS_ROOT}/test/iluvatar" gpu)
    else()
        add_subdirectory("${PHXFS_ROOT}/test/cuda" gpu)
    endif()
else()
    foreach(name test_batch_sync test_batch_async)
        add_executable(${name} "${PHXFS_ROOT}/test/${name}.cpp")
        target_link_libraries(${name} PRIVATE libphoenix Threads::Threads)
    endforeach()
    enable_testing()
    add_test(NAME batch_sync_cpu COMMAND test_batch_sync)
    add_test(NAME batch_async_cpu COMMAND test_batch_async)
endif()
